public VoxelShape makeShape(){
	VoxelShape shape = Shapes.empty();
	shape = Shapes.join(shape, Shapes.box(-0.5625, -0.6875, -0.5625, 1.5625, -0.375, 1.5625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-0.875, -1, -0.875, 1.875, -0.6875, 1.875), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-0.3125, -0.375, -0.3125, 1.3125, 0.125, 1.3125), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-0.5, 0.125, -0.5, 1.5, 0.625, 1.5), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-0.875, 0.625, -0.875, 1.875, 1.0625, 1.875), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-1, 1.625, -1, 2, 2, -0.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-0.875, 1.375, -0.75, -0.75, 1.625, 1.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-1, 1, -0.75, -0.75, 1.375, 1.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(1.75, 1, -0.75, 2, 1.375, 1.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-1, 1.625, -0.75, -0.75, 2, 1.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-1, 1, 1.75, 2, 1.375, 2), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-1, 1.625, 1.75, 2, 2, 2), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-1, 1, -1, 2, 1.375, -0.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 0.8125, -1, 1, 1.625, -0.875), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 0.8125, 1.875, 1, 1.625, 2), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-1, 0.8125, 0, -0.875, 1.625, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(1.875, 0.8125, 0, 2, 1.625, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(1.75, 1.625, -0.75, 2, 2, 1.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(1.75, 1.375, -0.75, 1.875, 1.625, 1.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-0.875, 1.375, -0.875, 1.875, 1.625, -0.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(-0.875, 1.375, 1.75, 1.875, 1.625, 1.875), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(1.625, -0.3125, 1.625, -0.625, -0.686875, -0.625), BooleanOp.OR);

	return shape;
}
